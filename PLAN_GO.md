```makefile
	sqlc generate -f db/sqlc.yaml

	goose -dir $(MIGRATIONS) sqlite3 $(DB_PATH) up

	goose -dir $(MIGRATIONS) sqlite3 $(DB_PATH) down

migrate-status: ## Show migration status
	goose -dir $(MIGRATIONS) sqlite3 $(DB_PATH) status

css: ## Build Tailwind CSS
	tailwindcss -i $(TAILWIND_IN) -o $(TAILWIND_OUT) --minify

css-watch: ## Watch Tailwind CSS
	tailwindcss -i $(TAILWIND_IN) -o $(TAILWIND_OUT) --watch

build: generate css ## Full build
	go build -o bin/server ./cmd/server

run: build ## Build and run
	./bin/server

dev: ## Hot-reload dev server (air + Tailwind watch)
	air

test: ## Run tests
	go test ./... -v -count=1

lint: ## Run linter
	golangci-lint run ./...

ingest: ## Run full historical data ingestion
	go run ./scripts/full_ingest.go

seed-parties: ## Seed party data from JSON
	go run ./scripts/seed_parties.go

clean: ## Remove generated files and DB
	rm -f bin/server $(TAILWIND_OUT) data/congress.db
	rm -f internal/database/models.go internal/database/*.sql.go
```

### `db/migrations/001_initial_schema.sql`

```sql
-- +goose Up

CREATE TABLE chambers (
    id   TEXT PRIMARY KEY,
    name TEXT NOT NULL
);

CREATE TABLE parties (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    name       TEXT NOT NULL UNIQUE,
    short_name TEXT,
    color      TEXT NOT NULL DEFAULT '#808080'
);

CREATE TABLE representatives (
    id               INTEGER PRIMARY KEY,
    external_id      INTEGER,
    chamber_id       TEXT    NOT NULL REFERENCES chambers(id),
    first_name       TEXT    NOT NULL,
    last_name        TEXT    NOT NULL,
    second_last_name TEXT,
    gender           TEXT,
    birth_date       TEXT,
    photo_url        TEXT,
    district         TEXT,
    region           TEXT
);

CREATE TABLE representative_parties (
    id                INTEGER PRIMARY KEY AUTOINCREMENT,
    representative_id INTEGER NOT NULL REFERENCES representatives(id),
    party_id          INTEGER NOT NULL REFERENCES parties(id),
    start_date        TEXT    NOT NULL,
    end_date          TEXT,
    UNIQUE(representative_id, party_id, start_date)
);

CREATE TABLE legislative_periods (
    id          INTEGER PRIMARY KEY,
    external_id INTEGER,
    start_date  TEXT,
    end_date    TEXT
);

CREATE TABLE legislatures (
    id                    INTEGER PRIMARY KEY,
    external_id           INTEGER,
    legislative_period_id INTEGER REFERENCES legislative_periods(id),
    start_date            TEXT,
    end_date              TEXT
);

CREATE TABLE sessions (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    external_id     INTEGER,
    chamber_id      TEXT    NOT NULL REFERENCES chambers(id),
    legislature_id  INTEGER REFERENCES legislatures(id),
    session_number  INTEGER,
    session_type    TEXT,
    date            TEXT    NOT NULL
);

CREATE TABLE bills (
    id          TEXT PRIMARY KEY,
    title       TEXT NOT NULL,
    description TEXT,
    law_number  TEXT,
    status      TEXT,
    filed_date  TEXT
);

CREATE TABLE votations (
    id               INTEGER PRIMARY KEY AUTOINCREMENT,
    external_id      INTEGER UNIQUE,
    chamber_id       TEXT    NOT NULL REFERENCES chambers(id),
    session_id       INTEGER REFERENCES sessions(id),
    bill_id          TEXT    REFERENCES bills(id),
    date             TEXT    NOT NULL,
    subject          TEXT,
    vote_type        TEXT,
    result           TEXT,
    quorum_type      TEXT,
    legislative_step TEXT,
    total_yes        INTEGER NOT NULL DEFAULT 0,
    total_no         INTEGER NOT NULL DEFAULT 0,
    total_abstain    INTEGER NOT NULL DEFAULT 0,
    total_absent     INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE individual_votes (
    id                INTEGER PRIMARY KEY AUTOINCREMENT,
    votation_id       INTEGER NOT NULL REFERENCES votations(id),
    representative_id INTEGER NOT NULL REFERENCES representatives(id),
    vote              TEXT    NOT NULL CHECK (vote IN ('yes','no','abstain','absent')),
    UNIQUE(votation_id, representative_id)
);

-- Indexes
CREATE INDEX idx_iv_votation       ON individual_votes(votation_id);
CREATE INDEX idx_iv_representative ON individual_votes(representative_id);
CREATE INDEX idx_iv_vote           ON individual_votes(vote);
CREATE INDEX idx_votations_chamber ON votations(chamber_id);
CREATE INDEX idx_votations_date    ON votations(date);
CREATE INDEX idx_votations_bill    ON votations(bill_id);
CREATE INDEX idx_reps_chamber      ON representatives(chamber_id);
CREATE INDEX idx_rep_parties_rep   ON representative_parties(representative_id);

-- +goose Down
DROP TABLE IF EXISTS individual_votes;
DROP TABLE IF EXISTS votations;
DROP TABLE IF EXISTS bills;
DROP TABLE IF EXISTS sessions;
DROP TABLE IF EXISTS legislatures;
DROP TABLE IF EXISTS legislative_periods;
DROP TABLE IF EXISTS representative_parties;
DROP TABLE IF EXISTS representatives;
DROP TABLE IF EXISTS parties;
DROP TABLE IF EXISTS chambers;
```

## 6. Example SQL Query Files

### `db/queries/votations.sql`

```sql
-- name: ListVotations :many
SELECT id, external_id, chamber_id, date, subject, vote_type, result,
       total_yes, total_no, total_abstain, total_absent
FROM   votations
WHERE  chamber_id = COALESCE(sqlc.narg('chamber_id'), chamber_id)
ORDER  BY date DESC
LIMIT  sqlc.arg('page_size') OFFSET sqlc.arg('page_offset');

-- name: CountVotations :one
SELECT COUNT(*) AS total
FROM   votations
WHERE  chamber_id = COALESCE(sqlc.narg('chamber_id'), chamber_id);

-- name: GetVotation :one
SELECT id, external_id, chamber_id, session_id, bill_id, date,
       subject, vote_type, result, quorum_type, legislative_step,
       total_yes, total_no, total_abstain, total_absent
FROM   votations
WHERE  id = sqlc.arg('id');

-- name: GetVotationByExternalID :one
SELECT id FROM votations WHERE external_id = sqlc.arg('external_id');

-- name: InsertVotation :one
INSERT INTO votations (external_id, chamber_id, session_id, bill_id, date,
                       subject, vote_type, result, quorum_type, legislative_step,
                       total_yes, total_no, total_abstain, total_absent)
VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
RETURNING id;
```

### `db/queries/individual_votes.sql`

```sql
-- name: InsertVote :exec
INSERT OR IGNORE INTO individual_votes (votation_id, representative_id, vote)
VALUES (?, ?, ?);

-- name: GetVotesByVotation :many
SELECT iv.vote,
       r.id AS representative_id,
       r.first_name,
       r.last_name,
       p.name       AS party_name,
       p.short_name AS party_short,
       p.color      AS party_color
FROM   individual_votes iv
JOIN   representatives r ON r.id = iv.representative_id
LEFT JOIN representative_parties rp
       ON rp.representative_id = r.id
      AND rp.start_date <= (SELECT date FROM votations WHERE id = iv.votation_id)
      AND (rp.end_date IS NULL
           OR rp.end_date >= (SELECT date FROM votations WHERE id = iv.votation_id))
LEFT JOIN parties p ON p.id = rp.party_id
WHERE  iv.votation_id = sqlc.arg('votation_id')
ORDER  BY iv.vote, r.last_name;
```

### `db/queries/analytics.sql`

```sql
-- name: GetPartyBreakdown :many
SELECT p.name       AS party_name,
       p.short_name AS party_short,
       p.color      AS party_color,
       iv.vote,
       COUNT(*)     AS count
FROM   individual_votes iv
JOIN   representatives r ON r.id = iv.representative_id
JOIN   representative_parties rp
       ON rp.representative_id = r.id
      AND rp.start_date <= (SELECT date FROM votations WHERE id = iv.votation_id)
      AND (rp.end_date IS NULL
           OR rp.end_date >= (SELECT date FROM votations WHERE id = iv.votation_id))
JOIN   parties p ON p.id = rp.party_id
WHERE  iv.votation_id = sqlc.arg('votation_id')
GROUP  BY p.id, p.name, p.short_name, p.color, iv.vote
ORDER  BY p.name, iv.vote;

-- name: GetRepresentativeVotingHistory :many
SELECT v.date,
       v.subject,
       b.title   AS bill_title,
       iv.vote,
       v.result
FROM   individual_votes iv
JOIN   votations v ON v.id = iv.votation_id
LEFT JOIN bills b  ON b.id = v.bill_id
WHERE  iv.representative_id = sqlc.arg('representative_id')
ORDER  BY v.date DESC
LIMIT  sqlc.arg('page_size') OFFSET sqlc.arg('page_offset');

-- name: GetPartyCohesion :many
SELECT p.name       AS party_name,
       p.color      AS party_color,
       v.id         AS votation_id,
       iv.vote      AS majority_vote,
       COUNT(*)     AS vote_count,
       SUM(COUNT(*)) OVER (PARTITION BY p.id, v.id) AS total_party_votes
FROM   individual_votes iv
JOIN   representatives r ON r.id = iv.representative_id
JOIN   representative_parties rp
       ON rp.representative_id = r.id
      AND rp.start_date <= v.date
      AND (rp.end_date IS NULL OR rp.end_date >= v.date)
JOIN   parties p   ON p.id = rp.party_id
JOIN   votations v ON v.id = iv.votation_id
WHERE  iv.vote IN ('yes', 'no')
GROUP  BY p.id, v.id, iv.vote
ORDER  BY p.name;
```

## 7. Key Code Patterns

### 7.1 Entry point — `cmd/server/main.go`

```go
package main

import (
    "context"
    "log/slog"
    "net/http"
    "os"
    "os/signal"
    "time"

    "congress-visualizer/internal/config"
    "congress-visualizer/internal/database"
    "congress-visualizer/internal/handlers"
    "congress-visualizer/internal/ingestion"
)

func main() {
    cfg := config.Load()

    db, err := database.Open(cfg.DBPath)
    if err != nil {
        slog.Error("failed to open database", "error", err)
        os.Exit(1)
    }
    defer db.Close()

    if err := database.Migrate(db, cfg.MigrationsDir); err != nil {
        slog.Error("failed to run migrations", "error", err)
        os.Exit(1)
    }

    queries := database.New(db)
    router := handlers.NewRouter(queries, cfg)

    // Start background sync
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()
    go ingestion.StartScheduler(ctx, queries, cfg)

    srv := &http.Server{
        Addr:         ":" + cfg.Port,
        Handler:      router,
        ReadTimeout:  5 * time.Second,
        WriteTimeout: 10 * time.Second,
    }

    // Graceful shutdown
    go func() {
        sigCh := make(chan os.Signal, 1)
        signal.Notify(sigCh, os.Interrupt)
        <-sigCh
        slog.Info("shutting down...")
        cancel()
        srv.Shutdown(context.Background())
    }()

    slog.Info("server starting", "port", cfg.Port)
    if err := srv.ListenAndServe(); err != http.ErrServerClosed {
        slog.Error("server error", "error", err)
    }
}
```

### 7.2 Router — `internal/handlers/router.go`

```go
package handlers

import (
    "net/http"

    "congress-visualizer/internal/config"
    "congress-visualizer/internal/database"
    "congress-visualizer/internal/render"

    "github.com/go-chi/chi/v5"
    "github.com/go-chi/chi/v5/middleware"
)

func NewRouter(q *database.Queries, cfg *config.Config) http.Handler {
    r := chi.NewRouter()
    r.Use(middleware.Logger)
    r.Use(middleware.Recoverer)
    r.Use(middleware.Compress(5))

    renderer := render.New("web/templates")

    // Static files
    fs := http.FileServer(http.Dir("web/static"))
    r.Handle("/static/*", http.StripPrefix("/static/", fs))

    // Pages
    r.Get("/", HomeHandler(q, renderer))
    r.Get("/votations", VotationListHandler(q, renderer))
    r.Get("/votations/{id}", VotationDetailHandler(q, renderer))
    r.Get("/representatives", RepresentativeListHandler(q, renderer))
    r.Get("/representatives/{id}", RepresentativeDetailHandler(q, renderer))
    r.Get("/parties", PartyListHandler(q, renderer))
    r.Get("/parties/{id}", PartyDetailHandler(q, renderer))
    r.Get("/bills", BillListHandler(q, renderer))
    r.Get("/bills/{id}", BillDetailHandler(q, renderer))

    // HTMX partials
    r.Get("/partials/vote-breakdown/{votationId}", VoteBreakdownPartial(q, renderer))
    r.Get("/partials/party-chart/{votationId}", PartyChartPartial(q, renderer))

    // Chart endpoints (return SVG directly)
```

### 7.5 Template rendering — `internal/render/render.go`

```go
package render

import (
    "html/template"
    "io"
    "net/http"
    "path/filepath"
)

type Renderer struct {
    templates *template.Template
}

func New(templateDir string) *Renderer {
    funcMap := template.FuncMap{
        "formatDate": FormatDate,
        "percentage": Percentage,
        "safeHTML":   func(s string) template.HTML { return template.HTML(s) },
    }

    tmpl := template.Must(
        template.New("").Funcs(funcMap).ParseGlob(
            filepath.Join(templateDir, "**", "*.html"),
        ),
    )
    return &Renderer{templates: tmpl}
}

func (r *Renderer) Page(w http.ResponseWriter, name string, data any) {
    w.Header().Set("Content-Type", "text/html; charset=utf-8")
    if err := r.templates.ExecuteTemplate(w, name, data); err != nil {
        http.Error(w, "render error", http.StatusInternalServerError)
    }
}

func (r *Renderer) Partial(w http.ResponseWriter, name string, data any) {
    // Same as Page but without the layout wrapper — used for HTMX swaps
    w.Header().Set("Content-Type", "text/html; charset=utf-8")
    if err := r.templates.ExecuteTemplate(w, name, data); err != nil {
        http.Error(w, "render error", http.StatusInternalServerError)
    }
}
```
